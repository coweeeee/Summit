<script>
const ratings = {};

function switchTab(name, btn) {
  document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
  document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
  document.getElementById('tab-' + name).classList.add('active');
  btn.classList.add('active');
  document.querySelector('.content').scrollTop = 0;
}

function toggleLike(btn) {
  const isLiked = btn.classList.toggle('liked');
  const num = parseInt(btn.textContent.trim().replace(/\D/g,''));
  const newNum = isLiked ? num + 1 : num - 1;
  btn.innerHTML = btn.innerHTML.replace(/\d+/, newNum);
}

function setRating(dim, val) {
  ratings[dim] = val;
  const row = document.querySelector(`.star-row[data-dim="${dim}"]`);
  row.querySelectorAll('.star').forEach((s, i) => {
    s.classList.toggle('on', i < val);
  });
  updateOverall();
}

function updateOverall() {
  const vals = Object.values(ratings);
  if (vals.length === 0) { document.getElementById('overall-score').textContent = '—'; return; }
  const avg = vals.reduce((a,b) => a+b, 0) / vals.length;
  document.getElementById('overall-score').textContent = avg.toFixed(1);
}

function filterClick(chip) {
  document.querySelectorAll('.filter-chip').forEach(c => c.classList.remove('active'));
  chip.classList.add('active');
}

function showToast(msg) {
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.classList.add('show');
  setTimeout(() => t.classList.remove('show'), 2800);
}

function submitHike() {
  showToast('Hike logged! Great adventure.');
  Object.keys(ratings).forEach(k => delete ratings[k]);
  document.querySelectorAll('.star').forEach(s => s.classList.remove('on'));
  document.getElementById('overall-score').textContent = '—';
  document.querySelectorAll('.form-input').forEach(i => { i.value = ''; });
  setTimeout(() => switchTab('profile', document.querySelectorAll('.nav-btn')[3]), 800);
}
